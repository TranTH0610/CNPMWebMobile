## packages/visitor-core/src/data/heritage.ts

Ghép bài với MP3 chỉ khi body khớp, kiểm tra khóa audio; không phụ thuộc UI.

Đường dẫn đầy đủ: `D:\3.HTML\Đồ án CNPM\ThuyetMinhTuDong\packages\visitor-core\src\data\heritage.ts`.

```typescript
/**
 * FILE NÀY LÀM GÌ: Ghép bài lịch sử có nguồn với MP3 đọc đúng văn bản.
 * mock.ts, Admin và bộ test dùng; phụ thuộc heritageContent.json/narrations.json.
 * UC02–UC06, UC14: nguồn chung cho cả bốn giao diện, không nhân bản bài trong UI.
 */
import raw from './heritageContent.json'
import recordings from './narrations.json'
import type { ContentSource, Language, PublishedText } from '../domain/model'

type HeritageEntry = {
  id: string; period: string; sources: ContentSource[]
  content: Record<Language, { address: string; summary: string; body: string }>
}
export const HERITAGE_CONTENT: readonly HeritageEntry[] = raw
export const NARRATIONS = recordings

// CHỨC NĂNG: Lấy MP3 đóng gói chỉ khi body trùng đúng bài đã tạo âm thanh.
// XỬ LÝ: Văn bản biên tập khác dù cùng ID/ngôn ngữ sẽ không mượn MP3 cũ.
export function bundledNarration(id: string, language: Language, body: string) {
  const entry = HERITAGE_CONTENT.find(place => place.id === id)
  if (entry?.content[language].body !== body) return undefined
  const recording = NARRATIONS.find(audio => audio.placeId === id && audio.language === language)
  return recording ? 'narration:' + recording.key : undefined
}

// CHỨC NĂNG: Chuyển khóa an toàn thành đường dẫn asset hoặc API.
// XỬ LÝ: Chỉ nhận SHA-256 đủ 64 ký tự; không ghép đường dẫn tùy ý từ QR/input.
export function narrationKey(url: string): string | undefined {
  return /^narration:[a-f0-9]{64}$/.test(url) ? url.slice(10) : undefined
}
export function bundledNarrationKey(key: string): boolean {
  return NARRATIONS.some(audio => audio.key === key)
}
export function heritageText(id: string, language: Language, title: string): PublishedText {
  const entry = HERITAGE_CONTENT.find(place => place.id === id)
  if (!entry) throw new Error('Missing heritage content: ' + id)
  const text = entry.content[language]
  const url = bundledNarration(id, language, text.body)
  if (!url) throw new Error('Missing narration: ' + id + '/' + language)
  const date = '2026-10-24T00:00:00Z'
  return {
    ...text, title, sources: entry.sources, adaptation: 'rewritten-and-translated',
    version: 2, status: 'published', approvedAt: date, publishedAt: date,
    audio: { id: url, url, version: 2, status: 'published', approvedAt: date, publishedAt: date },
  }
}
/* CÁCH KIỂM TRA: Có 8 bài × 5 ngôn ngữ; body thay đổi trả undefined;
 * mỗi audio có khóa khác nhau và đường dẫn không chứa tên máy/IP. */
```

```typescript
 * FILE NÀY LÀM GÌ:
 * Định nghĩa dữ liệu dùng chung cho UC01–UC11.
 * Service, hooks và các màn mobile/web sử dụng.
 * Không phụ thuộc React Native hoặc trình duyệt.
 */

// CHỨC NĂNG: Khai báo các ngôn ngữ được hỗ trợ.
// XỬ LÝ: Giữ cn làm mã nội bộ để tương thích lựa chọn đã lưu.
export const LANGUAGES = [
  { code: 'vi', label: 'Tiếng Việt' },
  { code: 'en', label: 'English' },
  { code: 'cn', label: '中文' },
  { code: 'pt', label: 'Português' },
  { code: 'ja', label: '日本語' },
] as const

export type Language = (typeof LANGUAGES)[number]['code']

// CHỨC NĂNG: Nhận diện ngôn ngữ từ bộ nhớ hoặc thiết bị.
// XỬ LÝ: zh-CN thành cn, pt_BR thành pt; không hỗ trợ trả null.
export function languageOf(
  value: string | null | undefined,
): Language | null {
  const base = value
    ?.toLowerCase()
    .replace(/_/g, '-')
    .split('-')[0]

  return base === 'zh'
    ? 'cn'
    : LANGUAGES.find((item) => item.code === base)?.code ?? null
}

// CHỨC NĂNG: Khai báo mã danh mục ổn định.
// XỬ LÝ: UI lấy tên qua i18n; không dùng tên tiếng Việt làm mã.
export const CATEGORIES = [
  'palace',
  'gate',
  'temple',
  'tomb',
  'museum',
  'monument',
] as const

export type Category = (typeof CATEGORIES)[number]

export type Zone = 'fortifications' | 'ceremonial'
export type SortOrder = 'name' | 'nearest'

export type Point = {
  lat: number
  lng: number
}

// CHỨC NĂNG: Mô tả một bản nội dung theo ngôn ngữ.
// XỬ LÝ: Service kiểm tra trạng thái duyệt/công bố trước khi trả cho khách.
// CHỨC NĂNG: Nguồn audio có phiên bản và trạng thái công bố riêng.
// XỬ LÝ: Optional trong PublishedText để dữ liệu UC01–UC11 cũ vẫn dùng được.
export type PublishedAudio = {
  id: string; url: string; version: number
  status: 'draft' | 'published'; approvedAt?: string; publishedAt?: string
}
// CHỨC NĂNG: Ghi nguồn để người đọc/Admin đối chiếu bài đã biên soạn.
// XỬ LÝ: Optional trong bản nội dung; không ép dữ liệu cũ phải có nguồn.
export type ContentSource = {
  title: string; url: string; accessedAt: string; license: string; licenseUrl: string
}
export type PublishedText = {
  sources?: readonly ContentSource[]
  adaptation?: 'rewritten-and-translated'

  audio?: PublishedAudio
  title: string
  summary: string
  body: string
  address: string
  version: number
  status: 'draft' | 'approved' | 'published'
  approvedAt?: string
  publishedAt?: string
}

// CHỨC NĂNG: Mô tả một địa điểm trong danh mục triển khai.
// XỬ LÝ: Các trường mới optional để dữ liệu cũ vẫn dùng được.
export type PlaceRecord = {
  id: string
  scope: string
  inLaunchCatalog: boolean
  status: 'draft' | 'published' | 'withdrawn'
  canonicalTitle: string
  zone: Zone

  period?: string
  category?: Category
  featured?: boolean

  // CHỨC NĂNG: Phân biệt tọa độ đã xác nhận.
  // XỬ LÝ: Chỉ dùng verified=true để tính khoảng cách.
  position?: Point & {
    verified: boolean
  }

  // CHỨC NĂNG: Cho phép thiếu một số bản ngôn ngữ.
  // XỬ LÝ: Không tự dịch chưa duyệt để lấp bản thiếu.
  content: Partial<Record<Language, PublishedText>>
}

// CHỨC NĂNG: Đọc danh mục của dữ liệu mới hoặc dữ liệu cũ.
// XỬ LÝ: Thiếu category thì suy ra từ zone để tương thích ngược.
export function categoryOf(place: PlaceRecord): Category {
  return place.category
    ?? (place.zone === 'fortifications' ? 'gate' : 'palace')
}

export type CatalogSource =
  () => Promise<readonly PlaceRecord[]>

// CHỨC NĂNG: Tách logic ngôn ngữ khỏi công nghệ lưu trữ.
// XỬ LÝ: Mobile cung cấp AsyncStorage, web cung cấp localStorage.

  readLanguage: () => Promise<string | null>
  writeLanguage: (language: Language) => Promise<void>
}
```

```typescript
/**
 * FILE NÀY LÀM GÌ: Kết nối audio/GPS/bộ nhớ riêng nền tảng vào JourneyProvider.
 * Vị trí: apps/mobile/src/services/journeyAdapters.ts. Liên quan UC01–UC11.
 * Phụ thuộc các import bên dưới; nghiệp vụ chung nằm trong visitor-core.
 */

import { narrationKey } from '@travelguide/visitor-core'
import { narrationAssets } from './narrationAssets'
import AsyncStorage from '@react-native-async-storage/async-storage'
import { AppState } from 'react-native'
import * as Location from 'expo-location'
import { createAudioPlayer, setAudioModeAsync } from 'expo-audio'
import type { JourneyAdapters } from '@travelguide/visitor-core/journey/react'
import type { Track } from '@travelguide/visitor-core/journey'
import vi from '../../assets/demo-audio/vi.mp3'
import en from '../../assets/demo-audio/en.mp3'
import cn from '../../assets/demo-audio/cn.mp3'
import pt from '../../assets/demo-audio/pt.mp3'
import ja from '../../assets/demo-audio/ja.mp3'

const demo: Record<string, number> = { vi, en, cn, pt, ja }

export const journeyAdapters: JourneyAdapters = {
  read: () => AsyncStorage.getItem('visitor-journey-v1'),
  write: value => value === null ? AsyncStorage.removeItem('visitor-journey-v1') : AsyncStorage.setItem('visitor-journey-v1', value),
  async watch(next, fail) {
    const permission = await Location.requestForegroundPermissionsAsync()
    if (permission.status !== 'granted') throw new Error('GPS denied')
    let stopped = false, generation = 0
    let subscription: Location.LocationSubscription | undefined
    const start = async () => {
      const request = ++generation
      try {
        const result = await Location.watchPositionAsync({ accuracy: Location.Accuracy.High, timeInterval: 3000, distanceInterval: 5 }, p => {
          if (!stopped && request === generation) next({ lat: p.coords.latitude, lng: p.coords.longitude,
            accuracy: p.coords.accuracy ?? Infinity, timestamp: p.timestamp })
        })
        if (stopped || request !== generation) result.remove(); else subscription = result
      } catch { if (!stopped) fail() }
    }
    const listener = AppState.addEventListener('change', state => {
      generation++; subscription?.remove(); subscription = undefined
      if (state === 'active' && !stopped) void start()
    })
    if (AppState.currentState === 'active') await start()
    return () => { stopped = true; generation++; listener.remove(); subscription?.remove() }
  },
  async open(url, title): Promise<Track> {
    await setAudioModeAsync({ playsInSilentMode: true, shouldPlayInBackground: true, interruptionMode: 'doNotMix' })
    // CHỨC NĂNG: Đọc asset cùng bài với web; bản Admin vừa tạo tải từ API RAM.
    // XỬ LÝ: Không dùng localhost của điện thoại; URL lấy từ cấu hình nền tảng.
    const key = narrationKey(url)
    const admin = process.env.EXPO_PUBLIC_APP_TARGET === 'admin'
    const base = admin ? process.env.EXPO_PUBLIC_ADMIN_API_URL : process.env.EXPO_PUBLIC_CATALOG_API_URL
    if (url.startsWith('narration:') && !key) throw new Error('Invalid narration key')
    const source = key ? narrationAssets[key] ?? (base ? base.replace(/\/$/, '') + '/audio/' + (admin ? 'preview/' : 'published/') + key + '.mp3' : undefined)
      : url.startsWith('demo:') ? demo[url.slice(5)] : url
    if (!source) throw new Error('Missing audio')
    const player = createAudioPlayer(source, { updateInterval: 250 })
    try {
      await new Promise<void>((resolve, reject) => {
        const timer = setTimeout(() => { listener.remove(); reject(new Error('Audio timeout')) }, 20000)
        const listener = player.addListener('playbackStatusUpdate', status => {
          if (status.error || status.isLoaded) { clearTimeout(timer); listener.remove(); if (status.error) reject(new Error(status.error)); else resolve() }
        })
        if (player.isLoaded) { clearTimeout(timer); listener.remove(); resolve() }
      })
    } catch (error) { player.remove(); throw error }
    let disposed = false
    return {
      duration: player.duration,
      play: async () => { player.setActiveForLockScreen(true, { title }); player.play() },
      pause: () => player.pause(), seek: seconds => player.seekTo(seconds), rate: value => player.setPlaybackRate(value),
      listen: callback => {
        const listener = player.addListener('playbackStatusUpdate', s => callback({ position: s.currentTime,
          duration: s.duration, playing: s.playing, ended: s.didJustFinish, error: Boolean(s.error) }))
        return () => listener.remove()
      },
      dispose: () => { if (!disposed) { disposed = true; player.setActiveForLockScreen(false); player.remove() } },
    }
  },
}
```

```typescript
 * Journey adapter và trình nghe thử Admin dùng; phụ thuộc visitor-core.
 * UC06/UC14.2: bản nháp chỉ qua API preview của site quản lý.
 */
import { bundledNarrationKey, narrationKey } from '@travelguide/visitor-core'
export function narrationSource(url: string, preview = false): string {
  const key = narrationKey(url)
  // CHỨC NĂNG: Audio sẵn có chạy ngay từ static, không cần gọi TTS lúc nghe.
  if (key) {
    if (bundledNarrationKey(key)) return '/narrations/' + key + '.mp3'
    const base: string = (preview ? import.meta.env.VITE_ADMIN_API_URL : import.meta.env.VITE_CATALOG_API_URL) || '/api/catalog'
    return base.replace(/\/$/, '') + '/audio/' + (preview ? 'preview/' : 'published/') + key + '.mp3'
  }
  if (url.startsWith('narration:')) throw new Error('Invalid narration key')
  return url.startsWith('demo:') ? '/demo-audio/' + url.slice(5) + '.mp3' : url
}
```

```typescript
/**
 * FILE NÀY LÀM GÌ: Kết nối audio/GPS/bộ nhớ riêng nền tảng vào JourneyProvider.
 * Vị trí: apps/web/src/services/journeyAdapters.ts. Liên quan UC01–UC11.
 * Phụ thuộc các import bên dưới; nghiệp vụ chung nằm trong visitor-core.
 */

import type { JourneyAdapters } from '@travelguide/visitor-core/journey/react'
import { narrationSource } from './narrationSource'
import type { Track } from '@travelguide/visitor-core/journey'

let mediaOwner: HTMLAudioElement | undefined

export const journeyAdapters: JourneyAdapters = {
  read: async () => localStorage.getItem('visitor-journey-v1'),
  write: async value => { if (value === null) localStorage.removeItem('visitor-journey-v1'); else localStorage.setItem('visitor-journey-v1', value) },
  async watch(next, fail) {
    if (!navigator.geolocation) throw new Error('GPS unavailable')
    const id = navigator.geolocation.watchPosition(p => next({ lat: p.coords.latitude, lng: p.coords.longitude,
      accuracy: p.coords.accuracy, timestamp: p.timestamp }), fail, { enableHighAccuracy: true, maximumAge: 5000, timeout: 15000 })
    return () => navigator.geolocation.clearWatch(id)
  },
  async open(url, title): Promise<Track> {
    const audio = new Audio(narrationSource(url))
    audio.preload = 'auto'
    try {
      // CHỨC NĂNG: Chờ metadata để thời lượng và thanh tua lấy từ file thật.
      // XỬ LÝ: Lỗi nguồn và timeout được trả về controller để cho thử lại.
      await new Promise<void>((resolve, reject) => {
        const done = () => { clearTimeout(timer); audio.removeEventListener('loadedmetadata', ready); audio.removeEventListener('error', bad) }
        const ready = () => { done(); resolve() }
        const bad = () => { done(); reject(new Error('Audio unavailable')) }
        const timer = setTimeout(bad, 20000)
        audio.addEventListener('loadedmetadata', ready)
        audio.addEventListener('error', bad)
        audio.load()
      })
    } catch (error) { audio.removeAttribute('src'); audio.load(); throw error }
    let disposed = false
    const track: Track = {
      duration: audio.duration,
      play: async () => {
        await audio.play()
        if ('mediaSession' in navigator) {
          mediaOwner = audio
          navigator.mediaSession.metadata = new MediaMetadata({ title })
          navigator.mediaSession.setActionHandler('play', () => { void audio.play().catch(() => audio.dispatchEvent(new Event('error'))) })
          navigator.mediaSession.setActionHandler('pause', () => audio.pause())
          navigator.mediaSession.setActionHandler('seekto', d => { if (d.seekTime !== undefined) audio.currentTime = d.seekTime })
        }
      },
      pause: () => audio.pause(),
      seek: async value => { audio.currentTime = value },
      rate: value => { audio.playbackRate = value },
      listen: callback => {
        const report = () => callback({ position: audio.currentTime, duration: audio.duration, playing: !audio.paused, ended: audio.ended, error: Boolean(audio.error) })
        const events = ['timeupdate', 'play', 'pause', 'ended', 'seeked', 'error']
        events.forEach(event => audio.addEventListener(event, report))
        return () => events.forEach(event => audio.removeEventListener(event, report))
      },
      dispose: () => {
        if (disposed) return
        disposed = true; audio.pause(); audio.removeAttribute('src'); audio.load()
        if ('mediaSession' in navigator && mediaOwner === audio) for (const action of ['play', 'pause', 'seekto'] as const) navigator.mediaSession.setActionHandler(action, null)
      },
    }
    return track
  },
}
```

```javascript
async function prepareNarration(body, language) {
  if (!['vi','en','cn','pt','ja'].includes(language) || !body.trim() || body.length > 20000) throw Error('Invalid narration')
  const key = createHash('sha256').update('gtts-v1|' + language + '|' + body).digest('hex')
  if (audioFile(key)) return 'narration:' + key
  if (jobs.has(key)) return jobs.get(key)

  // XỬ LÝ: Tệp request chỉ chứa bài công khai do Admin soạn; không truyền qua shell.
  if (jobs.size >= 2) throw Error('TTS busy')

    const folder = path.join(root, '.runtime/narrations')
    fs.mkdirSync(folder, {recursive:true})
    const request = path.join(folder, key + '.json'), output = path.join(folder, key + '.mp3')
    fs.writeFileSync(request, JSON.stringify({body,language}))
    try {
      await new Promise((resolve,reject) => execFile(process.env.NARRATION_PYTHON || 'python',
        [path.join(root,'scripts/generate-narrations.py'),'--request',request,'--output',output],
        {cwd:root,windowsHide:true,timeout:150000,env:{...process.env,PYTHONPATH:[path.join(root,'.tools/tts-python'),process.env.PYTHONPATH].filter(Boolean).join(path.delimiter)}},
        error => error ? reject(error) : resolve()))
      if (!audioFile(key)) throw Error('Missing MP3')
      return 'narration:' + key

  })()

  try { return await task } finally { jobs.delete(key) }
}

  res.setHeader('Content-Length',end-start+1);res.setHeader('Cache-Control','private, max-age=3600')

}
```

```python
    # CHỨC NĂNG: Đọc đúng văn bản được lưu, không thêm lời giới thiệu khác bài.
    # XỬ LÝ: Mỗi tác vụ có timeout và tối đa ba lần thử. Tệp tạm chỉ thay tệp
    # chính sau khi MP3 có thời lượng hợp lệ; mất mạng không để lại MP3 rỗng.
    code, tld = LANGUAGES[language]
    temporary = destination.with_suffix(".tmp.mp3")
    for attempt in range(3):
        try:
            gTTS(body, lang=code, tld=tld, timeout=(10, 35)).save(str(temporary))
            duration = MP3(temporary).info.length
            if duration < 1:
                raise ValueError("MP3 quá ngắn")
            temporary.replace(destination)
            return duration
        except Exception:
            temporary.unlink(missing_ok=True)
            if attempt == 2:
                raise
            time.sleep(attempt + 1)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--request", help="JSON {body,language} cho server RAM")
    parser.add_argument("--output", help="Tệp MP3 đích của server RAM")
    args = parser.parse_args()
    if args.request:
        request = json.loads(Path(args.request).read_text(encoding="utf-8"))
        destination = Path(args.output)
        destination.parent.mkdir(parents=True, exist_ok=True)
        synthesize(request["body"], request["language"], destination)
        return
    entries = json.loads((ROOT / "packages/visitor-core/src/data/heritageContent.json").read_text(encoding="utf-8"))
    web = ROOT / "apps/web/static/narrations"
    mobile = ROOT / "apps/mobile/assets/narrations"
    web.mkdir(parents=True, exist_ok=True)
    mobile.mkdir(parents=True, exist_ok=True)
    jobs = [(entry["id"], lang, text["body"]) for entry in entries for lang, text in entry["content"].items()]
    def generate(job):
        id, lang, body = job
        text_hash = hashlib.sha256(body.encode("utf-8")).hexdigest()
        key = hashlib.sha256(("gtts-v1|" + lang + "|" + body).encode("utf-8")).hexdigest()
        target = web / (key + ".mp3")
        duration = MP3(target).info.length if target.exists() and target.stat().st_size > 0 else synthesize(body, lang, target)
        (mobile / target.name).write_bytes(target.read_bytes())
        print(id, lang, round(duration, 2), "seconds", flush=True)
        return {"placeId": id, "language": lang, "key": key, "textSha256": text_hash, "audioSha256": hashlib.sha256(target.read_bytes()).hexdigest(), "duration": round(duration, 3), "bytes": target.stat().st_size, "provider": "gTTS", "speechLanguage": LANGUAGES[lang][0]}
    # CHỨC NĂNG: Giới hạn bốn kết nối; chỉ ghi manifest sau khi đủ 40 bài.
    with ThreadPoolExecutor(max_workers=4) as pool:
        records = list(pool.map(generate, jobs))
    (ROOT / "packages/visitor-core/src/data/narrations.json").write_text(json.dumps(records, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    lines = ["/** FILE NÀY LÀM GÌ: Ánh xạ MP3 tĩnh để Metro đóng gói đúng 40 bản đọc.", " * generate-narrations.py sinh; journeyAdapters và AudioPreview dùng. Không sửa tay. */"]
    for index, record in enumerate(records):
        lines.append(f"import audio{index} from '../../assets/narrations/{record['key']}.mp3'")
    lines.extend(["// CHỨC NĂNG: Tra bằng dấu vân tay của văn bản/ngôn ngữ, tránh phát nhầm bài.", "export const narrationAssets: Readonly<Record<string, number>> = {"])
    lines.extend(f"  '{record['key']}': audio{index}," for index, record in enumerate(records))
    lines.append("}\n")
    (ROOT / "apps/mobile/src/services/narrationAssets.ts").write_text("\n".join(lines), encoding="utf-8")

if __name__ == "__main__":
    main()
```
